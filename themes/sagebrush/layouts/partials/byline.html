{{- /* byline — renders a "By X" line, with a small circular avatar per author, for a page that has an explicit authors list (via frontmatter or section cascade). Pages without explicit authorship render nothing — the site-wide author is reserved for the footer and cite-this-page partial. Usage: {{ partial "byline.html" . }} Accepts either `authors: [Name, Name]` (preferred, list) or `author: Name` (singular shorthand) in frontmatter. Avatars come from data/authors.yaml, keyed by the exact author name; an author with no entry renders as name only. */ -}} {{- $authors := slice -}} {{- with .Params.authors -}} {{- $authors = . -}} {{- else -}} {{- with .Params.author }}{{ $authors = slice . }}{{ end -}} {{- end -}} {{- if $authors }} {{- $last := sub (len $authors) 1 }} By {{ range $i, $name := $authors -}} {{- if gt $i 0 }}{{ if eq $i $last }} and {{ else }}, {{ end }}{{ end -}} {{- with index hugo.Data.authors $name }}{{ with .avatar }}{{ end }}{{ end -}} {{ $name -}} {{- end }} {{- end -}}
